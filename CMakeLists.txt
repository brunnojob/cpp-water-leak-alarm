cmake_minimum_required(VERSION 3.20)
project(water_leak_monitor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(leak_monitor src/main.cpp)
target_include_directories(leak_monitor PRIVATE include)
target_compile_options(leak_monitor PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(leak_tests tests/leak_test.cpp)
target_include_directories(leak_tests PRIVATE include)
add_test(NAME leak COMMAND leak_tests)
